include_directories(${REACTOS_SOURCE_DIR}/sdk/include/reactos/subsys)
spec2def(wow64.dll wow64.spec ADD_IMPORTLIB)
add_library(wow64 MODULE init.c thread.c process.c syscall.c fastsvc.c ioctl.c scsi.c csr.c lpc.c exception.c environment.c redirection.c debug.c ${CMAKE_CURRENT_BINARY_DIR}/wow64.def)
# The plain services never touch the guest's x87/SSE registers: nothing is saved around them (cpu.S)
set_source_files_properties(fastsvc.c PROPERTIES COMPILE_FLAGS "-mgeneral-regs-only")
set_module_type(wow64 win32dll ENTRYPOINT 0)
target_link_libraries(wow64 libcntpr ${PSEH_LIB})
add_importlibs(wow64 wow64cpu wow64win ntdll)
add_dependencies(wow64 wow64_syscall_tables)
add_cd_file(TARGET wow64 DESTINATION reactos/system32 FOR all)
