; Not in the online RAPPS database, so featmgr stages this file into the RAPPS appdb right before it
; runs "rapps /install /S powershell-7". The default amd64 image carries the unmodified official MSI and
; featmgr installs it from there (the .ini's LocalInstaller=, no download); otherwise rapps downloads the official
; MSI from the PowerShell GitHub release at install time, checks its SHA-256 (and SHA1), and runs

;

;   ADD_PATH=1             add the install folder to the system PATH
;   ENABLE_PSREMOTING=0    do not enable PowerShell remoting (nitro has no WinRM)
;   REGISTER_MANIFEST=0    do not register the ETW/event manifest (wevtutil)
;   USE_MU=0 ENABLE_MU=0   no Microsoft Update opt-in
;   ADD_EXPLORER_CONTEXT_MENU_OPENPOWERSHELL=0 ADD_FILE_CONTEXT_MENU_RUNPOWERSHELL=0  no shell verbs
